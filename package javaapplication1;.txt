package javaapplication1;

public class calculateDistant {
    private double x_serv,y_serv;

    public calculateDistant(double x_serv, double y_serv) {
        this.x_serv = x_serv;
        
        this.y_serv = y_serv;
        
    }

    public double getX_serv1() {
        return x_serv;
    }

    public void setX_serv1(double x_serv1) {
        this.x_serv = x_serv1;
    }

   
    public double getY_serv() {
        return y_serv;
    }

    public void setY_serv(double y_serv) {
        this.y_serv = y_serv;
    }

   
public double getDistant(double x,double y){
    double dx = this.x_serv - x;
        double dy = this.y_serv - y;
        return Math.sqrt(Math.pow(dx, 2) + Math.pow(dy, 2));
}
    
}
